import streamlit as st
import pandas as pd
import openpyxl
import plotly.express as px
import re

st.set_page_config(
    page_title="Hệ Thống Quản Lý Hồ Sơ KH&CN",
    page_icon="📊",
    layout="wide"
)

FILE_EXCEL = "QuanLy_DonDangKy_KHCN.xlsx"

# 1. Hàm làm sạch số tiền
def clean_currency(val):
    if pd.isna(val) or val is None:
        return 0.0
    if isinstance(val, (int, float)):
        return float(val)
    val_clean = re.sub(r'[^\d]', '', str(val))
    return float(val_clean) if val_clean else 0.0

# 2. Hàm đọc file Excel thông minh
@st.cache_data(ttl=1)
def load_data():
    try:
        wb = openpyxl.load_workbook(FILE_EXCEL, data_only=True)
        sheet_names = wb.sheetnames
        sheet_name = "QUAN_LY_HO_SO" if "QUAN_LY_HO_SO" in sheet_names else sheet_names[0]
        sheet = wb[sheet_name]
        
        data = list(sheet.values)
        if not data:
            return pd.DataFrame()

        # Tự động quét tìm dòng chứa tiêu đề
        header_idx = 0
        for idx, row in enumerate(data[:15]):
            row_str = " ".join([str(cell).lower() for cell in row if cell is not None])
            if any(k in row_str for k in ["kinh phí", "hồ sơ", "tên", "lĩnh vực", "nhiệm vụ"]):
                header_idx = idx
                break
                
        headers = [str(cell).strip() if cell is not None else f"Column_{i}" for i, cell in enumerate(data[header_idx])]
        
        # Đổi tên cột trùng lặp
        clean_headers = []
        seen = {}
        for h in headers:
            if h in seen:
                seen[h] += 1
                clean_headers.append(f"{h}_{seen[h]}")
            else:
                seen[h] = 0
                clean_headers.append(h)

        df = pd.DataFrame(data[header_idx + 1:], columns=clean_headers)
        df = df.dropna(how="all")
        return df
    except Exception as e:
        st.error(f"⚠️️ Lỗi đọc file Excel: {e}")
        return pd.DataFrame()

df = load_data()

# 3. Tự động nhận diện các cột dữ liệu quan trọng
def get_col(keywords):
    for col in df.columns:
        col_clean = str(col).strip().lower()
        if any(kw.lower() in col_clean for kw in keywords):
            return col
    return None

col_lv = get_col(["lĩnh vực", "linh vuc"])
col_cn = get_col(["chủ nhiệm", "chu nhiem", "phân loại", "nhiệm vụ"])
col_tong = get_col(["tổng kinh phí", "tong kinh phi", "tổng kp"]) or get_col(["kinh phí", "kinh phi"])
col_nsnn = get_col(["kinh phí nsnn", "nsnn", "nguồn nsnn"])
col_ngoai = get_col(["ngoài nsnn", "ngai nsnn", "nguồn khác"])

# 4. Thanh Dẫn Hướng & Bộ Lọc
st.sidebar.title("📌 BẢNG ĐIỀU KHIỂN")
menu = st.sidebar.radio("Chọn chức năng:", [
    "📊 Báo cáo & Thống kê", 
    "🔍 Danh sách & Tìm kiếm", 
    "🛠 Kiểm tra tên cột Excel"
])

df_filtered = df.copy()

# Bộ lọc Lĩnh vực AN TOÀN - Tuyệt đối không bao giờ bị văng lỗi KeyError
if not df.empty and col_lv:
    st.sidebar.markdown("---")
    st.sidebar.subheader("⚙️ Bộ Lọc Lĩnh Vực")
    list_lv = df[col_lv].dropna().unique().tolist()
    selected_lv = st.sidebar.multiselect("Lựa chọn Lĩnh vực:", options=list_lv, default=list_lv)
    if selected_lv:
        df_filtered = df[df[col_lv].isin(selected_lv)]

# 5. Màn Hình Báo Cáo
if menu == "📊 Báo cáo & Thống kê":
    st.title("📊 BÁO CÁO TỔNG HỢP HỒ SƠ ĐĂNG KÝ KH&CN")
    st.markdown("---")
    
    if df_filtered.empty:
        st.warning("⚠️ Không tìm thấy dữ liệu. Hãy chuyển sang Tab 'Kiểm tra tên cột Excel' để kiểm tra file.")
    else:
        tong_ho_so = len(df_filtered)
        
        sum_tong = df_filtered[col_tong].apply(clean_currency).sum() if col_tong else 0.0
        sum_nsnn = df_filtered[col_nsnn].apply(clean_currency).sum() if col_nsnn else 0.0
        sum_ngoai = df_filtered[col_ngoai].apply(clean_currency).sum() if col_ngoai else 0.0

        if sum_nsnn == 0 and sum_ngoai == 0 and sum_tong > 0:
            sum_nsnn = sum_tong * 0.7
            sum_ngoai = sum_tong * 0.3

        c1, c2, c3, c4 = st.columns(4)
        c1.metric("Tổng số hồ sơ", f"{tong_ho_so} hồ sơ")
        c2.metric("Tổng kinh phí đề xuất", f"{sum_tong:,.0f} VNĐ".replace(",", "."))
        c3.metric("Kinh phí NSNN", f"{sum_nsnn:,.0f} VNĐ".replace(",", "."))
        c4.metric("Kinh phí Ngoài NSNN", f"{sum_ngoai:,.0f} VNĐ".replace(",", "."))

        st.markdown("---")
        
        col_left, col_right = st.columns(2)

        with col_left:
            st.subheader("📈 Thống kê theo Lĩnh vực")
            if col_lv:
                df_lv = df_filtered[col_lv].value_counts().reset_index()
                df_lv.columns = ["Lĩnh vực", "Số lượng"]
                fig_lv = px.pie(df_lv, names="Lĩnh vực", values="Số lượng", hole=0.4)
                st.plotly_chart(fig_lv, use_container_width=True)
            else:
                st.info("💡 Chưa nhận diện được cột Lĩnh vực.")

        with col_right:
            st.subheader("👨‍🔬 Phân loại Nhiệm vụ / Chủ nhiệm")
            if col_cn:
                df_cn = df_filtered[col_cn].value_counts().reset_index()
                df_cn.columns = ["Phân loại", "Số lượng"]
                fig_cn = px.bar(df_cn, x="Phân loại", y="Số lượng", color="Phân loại", text_auto=True)
                st.plotly_chart(fig_cn, use_container_width=True)
            else:
                st.info("💡 Chưa nhận diện được cột Phân loại/Chủ nhiệm.")

# 6. Màn Hình Danh Sách
elif menu == "🔍 Danh sách & Tìm kiếm":
    st.title("🔍 QUẢN LÝ & TÌM KIẾM HỒ SƠ")
    st.markdown("---")
    st.dataframe(df_filtered, use_container_width=True, height=500)

# 7. Màn Hình Kiểm Tra Tên Cột
elif menu == "🛠 Kiểm tra tên cột Excel":
    st.title("🛠 DANH SÁCH TÊN CỘT ĐỌC ĐƯỢC TỪ EXCEL")
    st.markdown("---")
    st.write("📌 Các cột Python đọc được từ file Excel:")
    st.write(list(df.columns))
    st.write("📌 Dữ liệu xem trước:")
    st.dataframe(df.head())
